<script>

const STORAGE_KEY = "LINCSS_PRESENTATION_WORKBOOK_V1";

const saveFields = () =>
  Array.from(document.querySelectorAll("[data-save]"));


function collectData() {

  const data = {
    savedAt: new Date().toISOString(),
    fields: {}
  };

  saveFields().forEach(field => {

    if (field.type === "checkbox") {
      data.fields[field.id] = field.checked;
    } else {
      data.fields[field.id] = field.value;
    }

  });

  return data;
}


function applyData(data) {

  if (!data || !data.fields) return;

  saveFields().forEach(field => {

    if (!(field.id in data.fields)) return;

    if (field.type === "checkbox") {
      field.checked = Boolean(data.fields[field.id]);
    } else {
      field.value = data.fields[field.id] ?? "";
    }

  });

  updateProgress();
}


function saveWorkbook(showMessage = true) {

  const data = collectData();

  localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify(data)
  );

  if (showMessage) {
    showSaveStatus("Saved on this device");
  }

  updateProgress();
}


function loadWorkbook() {

  const raw = localStorage.getItem(STORAGE_KEY);

  if (!raw) {
    updateProgress();
    return;
  }

  try {

    const data = JSON.parse(raw);

    applyData(data);

    showSaveStatus("Previous work restored");

  } catch (error) {

    console.error("Could not restore workbook.", error);

  }
}


let autosaveTimer = null;

function scheduleAutosave() {

  showSaveStatus("Saving…");

  clearTimeout(autosaveTimer);

  autosaveTimer = setTimeout(() => {
    saveWorkbook(false);
    showSaveStatus("Saved automatically");
  }, 500);

}


function showSaveStatus(message) {

  const el = document.getElementById("saveStatus");

  el.textContent = message;

  clearTimeout(el._timer);

  el._timer = setTimeout(() => {
    el.textContent = "Ready";
  }, 2500);

}


function updateProgress() {

  const fields = saveFields();

  let completed = 0;

  fields.forEach(field => {

    if (field.type === "checkbox") {

      if (field.checked) completed++;

    } else {

      if (field.value.trim().length > 0) completed++;

    }

  });

  const percent =
    fields.length === 0
      ? 0
      : Math.round((completed / fields.length) * 100);

  document.getElementById("progressNumber").textContent = percent;

  document.getElementById("progressBar").style.width =
    percent + "%";

}


function downloadBackup() {

  const data = collectData();

  const studentName =
    document.getElementById("studentName").value.trim()
    || "Student";

  const topic =
    document.getElementById("scienceTopic").value.trim()
    || "Science";

  const safeName =
    (studentName + "_" + topic)
      .replace(/[^a-z0-9_-]+/gi, "_");

  const blob = new Blob(
    [JSON.stringify(data, null, 2)],
    { type: "application/json" }
  );

  const url = URL.createObjectURL(blob);

  const a = document.createElement("a");

  a.href = url;

  a.download =
    "LINCSS_Workbook_Backup_" + safeName + ".json";

  document.body.appendChild(a);

  a.click();

  a.remove();

  URL.revokeObjectURL(url);

  showSaveStatus("Backup downloaded");
}


function importBackup(event) {

  const file = event.target.files[0];

  if (!file) return;

  const reader = new FileReader();

  reader.onload = function() {

    try {

      const data = JSON.parse(reader.result);

      applyData(data);

      saveWorkbook(false);

      showSaveStatus("Backup imported");

    } catch (error) {

      alert(
        "That file does not appear to be a valid LINCSS workbook backup."
      );

    }

  };

  reader.readAsText(file);

  event.target.value = "";
}


function clearWorkbook() {

  const confirmed = confirm(
    "Clear every response saved in this workbook on this device?\n\n" +
    "This cannot be undone unless you previously downloaded a backup."
  );

  if (!confirmed) return;

  localStorage.removeItem(STORAGE_KEY);

  saveFields().forEach(field => {

    if (field.type === "checkbox") {
      field.checked = false;
    } else {
      field.value = "";
    }

  });

  updateProgress();

  showSaveStatus("Workbook cleared");
}


saveFields().forEach(field => {

  field.addEventListener("input", scheduleAutosave);

  field.addEventListener("change", scheduleAutosave);

});


window.addEventListener("load", loadWorkbook);

</script>
